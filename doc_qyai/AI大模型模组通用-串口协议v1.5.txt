/*

1、AI模组下发串口协议：2字节头部 + 2字节和校验 + 1字数据类型 + 1字预留类型 +  2字节数据长度  + n字节数据
                |   2BYTE  |   2BYTE     |  1BYTE      |     1BYTE   |     2BYTE       | nBYTE ........  |
                |   0xA55A |    sum      |  datatype   |     resv    |     len         | data0 ... datan |
头部head = 0xA55A
和校验sum：datatype + resv + len + data-0-n求和，即：除了head和sum之外的所有数据求和
数据类型datatype：详见 enum datatype
预留类型resv：0
数据长度len: 数据长度
数据data：len个字节的data0 ... datan
注意：数据存放格式：小端模式


datatype 为数据类型则为：typedef enum datatype。
注意：当len = 0，则没有数据：则可能是只有表情，息屏，开始录音，结束录音。
data数据为：
	语音识别下发UTF8文本格式数据：	介绍你自己
	AI对话下发UTF8文本格式数据：  	我是豆包
	闹钟下发数据为json文本格式：  //alarm_cyc ： 0 是只设置1次；alarm_cyc ： 1是每天使能；后面这个暂时支持：alarm_cyc = 2是工作日；alarm_cyc = 3是周末
								{"alarm_cyc": 1,"alarm_time": "2025-04-10 20:08"}
	倒计时计时器下发文本格式（单位为秒）： {"count_down":100}
				
结构体
struct tx_head
{
	unsigned short head;//0xA55A
    unsigned short checksum;//和校验sum：datatype + len + data求和
    unsigned char datatype;//数据类型
    unsigned char resv;//预留：默认为0
    unsigned short len;//len数据的长度
	unsigned char data[0];//数据
};
注意：当单片机设备控制AI大模型模组时候，单片机发送的数据格式也为上述格式即可。


2、AI模组接收串口协议：2字节头部 + 1字节返回类型 + 1字节预留 （AI模组暂时不支持接收出错方式，此处为预留）
|   2BYTE  |   1BYTE    | 1BYTE |
|   ABAB   |  err_type  | 预留  |

返回err_type类型：
	0：成功接收数据
	1：接收数据校验失败（需要重发）
	...预留

结构体
struct rx_head
{
	unsigned short head;//0xABBA
	unsigned char err_type;//出错类型
	unsigned char resv;//预留
};
注意：当AI大模型模组发上述步骤1的数据给到单片机设备时，单片机接收出货则需要回复上述步骤2的协议。

3、当单片机设备控制AI模组时候，单片机发送的数据格式为上述的1的说明协议即可。


//-----------------------------------以下为C的例子----------------------------------------------//
//发送
#define SEND_HEAD   0xA55A
#pragma pack(push, 4) //务必4字节对齐
struct ai_tx_uart_info{//7字节
    unsigned short head;//0xA55A
    unsigned short checksum;//和校验sum：len + data求和
    unsigned char datatype;//数据类型
    unsigned char resv;//预留：默认为0
    unsigned short len;//len数据的长度
	unsigned char data[0];//数据
};
#pragma pack(pop)

//接收
#define RECV_HEAD   0xABBA
#pragma pack(push, 1)
struct ai_rx_uart_head
{
	unsigned short head;//0xABAB
	unsigned char err_type;//出错类型
	unsigned char resv;//预留
};
#pragma pack(pop)


//校验和
static unsigned short data_check_sum(unsigned char *data, int len)
{
    int i;
    unsigned short sum = 0;
    for(i = 0; i < len; i++){
        sum += data[i];
    }
    return sum;
}
static void uart_send_data(char *buf, int len)
{

}
int ai_uart_cmd_data_send(unsigned char data_type, char *buf, int len)//外文件使用的函数
{
    struct ai_tx_uart_info *tx_info = NULL;
	tx_info = malloc(sizeof(struct ai_tx_uart_info) + len);
	if (tx_info) {
		memset(tx_info, 0, sizeof(struct ai_tx_uart_info));
		tx_info->head = SEND_HEAD;
		tx_info->len = len;
		tx_info->datatype = data_type;
		if(buf && len){
			memcpy(tx_info->data, buf, len);
		}
		tx_info->checksum = data_check_sum(&tx_info->datatype, tx_info->len + sizeof(struct ai_tx_uart_info) - (sizeof(tx_info->head) - sizeof(tx_info->checksum)));;
		uart_send_data(tx_info, sizeof(struct ai_tx_uart_info) + tx_info->len);//串口发送
		free(tx_info);
		//printf("-> txt : %s \n",buf);
	}else{
		printf("-> alloc err\n");
	}
    return len;
}

int ai_uart_data_recv(char *buf, int len)//串口接收到的数据解析
{
    struct ai_rx_uart_head *rx_info = (struct ai_rx_uart_head *)buf;
    struct ai_tx_uart_info *tx_info = NULL;
    enum datatype type = 0;

    if(!rx_info){
        return -1;
    }
    if(rx_info->head == RECV_HEAD){
        type = rx_info->err_type;
        printf("rx_info->err_type = %d\n",type);
    }else if(rx_info->head == SEND_HEAD){//收到的AI大模型数据
        tx_info = (struct ai_tx_uart_info *)buf;
        type = tx_info->datatype;
		printf("buf = %s\n",buf);
        switch(type) {
            // AI对话相关命令
            case AI_UART_CMD_DIALUOGE_START:    // 48：启动AI对话
            case AI_UART_CMD_GET_VBAT:          //140：电量查询，带有数据则为电量值
            case AI_UART_CMD_GET_NET_CONNET:    //141：查看wifi联网状态
            case AI_UART_CMD_GET_BT_CONNET:     //142：查看蓝牙连接状态
            default:
                break;
        }
    }
    return 0;
